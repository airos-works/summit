cmake_minimum_required(VERSION 3.18)
project(Summit VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SUMMIT_MODERN_WEBKIT "Build Summit with the native multiprocess WebKit API" OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(summit_core src/core/Address.cpp src/core/Protocol.cpp src/core/Profile.cpp src/core/ExtensionCatalog.cpp src/core/ExtensionIdentity.cpp src/core/ExtensionStore.cpp
    src/core/Favicon.cpp src/core/InternalPages.cpp src/core/Zoom.cpp src/core/Suggest.cpp src/core/DevTools.cpp src/core/DevToolsFormat.cpp
    src/core/Mcp.cpp src/core/McpHttp.cpp src/core/McpInspector.cpp)
target_link_libraries(summit_core PUBLIC OpenSSL::Crypto)
target_include_directories(summit_core PUBLIC src vendor)
target_compile_options(summit_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(summit_tests tests/CoreTests.cpp)
target_link_libraries(summit_tests PRIVATE summit_core)
add_test(NAME summit_core COMMAND summit_tests)
add_executable(summit_devtools_tests tests/DevToolsTests.cpp)
target_link_libraries(summit_devtools_tests PRIVATE summit_core)
add_test(NAME summit_devtools COMMAND summit_devtools_tests)
find_package(Threads REQUIRED)
add_executable(summit_mcp_tests tests/McpTests.cpp)
target_link_libraries(summit_mcp_tests PRIVATE summit_core Threads::Threads)
add_test(NAME summit_mcp COMMAND summit_mcp_tests)
add_executable(summit_extension_catalog_tests tests/ExtensionCatalogTests.cpp)
target_link_libraries(summit_extension_catalog_tests PRIVATE summit_core)
add_test(NAME summit_extension_catalog COMMAND summit_extension_catalog_tests)
add_executable(summit_extension_identity_tests tests/ExtensionIdentityTests.cpp)
target_link_libraries(summit_extension_identity_tests PRIVATE summit_core)
add_test(NAME summit_extension_identity COMMAND summit_extension_identity_tests ${CMAKE_SOURCE_DIR}/tests/fixtures/extensions/chrome-key/public-key.txt)
add_executable(summit_extension_store_tests tests/ExtensionStoreTests.cpp)
target_link_libraries(summit_extension_store_tests PRIVATE summit_core)
add_test(NAME summit_extension_store COMMAND summit_extension_store_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Haiku")
    if(SUMMIT_MODERN_WEBKIT)
        find_path(WEBKIT_MODERN_INCLUDE_DIR WebKit/WebKitView.h PATH_SUFFIXES webkit REQUIRED)
        find_library(WEBKIT_MODERN_LIBRARY WebKit REQUIRED)
        set(SUMMIT_WEBKIT_INCLUDE_DIR ${WEBKIT_MODERN_INCLUDE_DIR})
        set(SUMMIT_WEBKIT_LIBRARY ${WEBKIT_MODERN_LIBRARY})
    else()
        find_path(WEBKIT_INCLUDE_DIR WebView.h PATH_SUFFIXES webkit webkit/haiku REQUIRED)
        find_library(WEBKIT_LIBRARY WebKitLegacy REQUIRED)
        set(SUMMIT_WEBKIT_INCLUDE_DIR ${WEBKIT_INCLUDE_DIR})
        set(SUMMIT_WEBKIT_LIBRARY ${WEBKIT_LIBRARY})
    endif()
    find_program(HAIKU_RC rc REQUIRED)
    find_program(HAIKU_XRES xres REQUIRED)
    add_executable(Summit src/main.cpp src/ui/BrowserWindow.cpp src/ui/Chrome.cpp src/ui/ExtensionPermissionPrompt.cpp src/ui/ExtensionController.cpp src/ui/ExtensionInstaller.cpp src/ui/ExtensionManager.cpp
        src/ui/ExtensionStoreDownload.cpp src/ui/CertificateInfoWindow.cpp src/ui/ReaderMode.cpp
        src/ui/ProtocolHandlers.cpp src/ui/FaviconCache.cpp src/ui/PreferencesWindow.cpp src/ui/DefaultBrowser.cpp src/ui/SharedProfile.cpp src/ui/DirectWindowRescue.cpp
        src/ui/AddressSuggestions.cpp src/ui/SitePermissions.cpp src/ui/Location.cpp
        src/ui/SourceView.cpp src/ui/DevToolsWindow.cpp src/ui/DevToolsNetwork.cpp src/ui/DevToolsConsole.cpp src/ui/DevToolsStorage.cpp
        src/ui/BrowserWindowMcp.cpp src/ui/McpService.cpp src/ui/McpLog.cpp src/ui/McpBrowserTools.cpp src/ui/McpSystemTools.cpp
        src/ui/McpPageTools.cpp src/ui/McpEngineTools.cpp src/ui/McpDebugHelper.cpp)
    target_include_directories(Summit PRIVATE ${SUMMIT_WEBKIT_INCLUDE_DIR} /boot/system/develop/headers/private/netservices)
    target_compile_options(Summit PRIVATE -Wno-multichar)
    target_compile_definitions(Summit PRIVATE SUMMIT_MODERN_WEBKIT=$<BOOL:${SUMMIT_MODERN_WEBKIT}>)
    if(SUMMIT_MODERN_WEBKIT)
        target_compile_features(Summit PRIVATE cxx_std_23)
        target_compile_definitions(Summit PRIVATE BUILDING_HAIKU__=1)
        # The developer tools: Scintilla and Lexilla show their text, as in Kiri.
        find_path(SCINTILLA_INCLUDE_DIR ScintillaView.h PATH_SUFFIXES scintilla REQUIRED)
        find_path(LEXILLA_INCLUDE_DIR SciLexer.h PATH_SUFFIXES lexilla REQUIRED)
        find_library(SCINTILLA_LIBRARY scintilla REQUIRED)
        find_library(LEXILLA_LIBRARY lexilla REQUIRED)
        target_include_directories(Summit PRIVATE ${SCINTILLA_INCLUDE_DIR} ${LEXILLA_INCLUDE_DIR} /boot/system/develop/headers/private)
        target_link_libraries(Summit PRIVATE ${SCINTILLA_LIBRARY} ${LEXILLA_LIBRARY} columnlistview)
    endif()
    target_link_libraries(Summit PRIVATE summit_core ${SUMMIT_WEBKIT_LIBRARY} be game network bnetapi translation tracker curl)
    set_property(TARGET Summit APPEND PROPERTY LINK_DEPENDS ${CMAKE_SOURCE_DIR}/resources/Summit.rdef)
    set_property(TARGET Summit APPEND PROPERTY LINK_DEPENDS ${CMAKE_SOURCE_DIR}/resources/Summit.hvif)
    file(GLOB SUMMIT_TOOLBAR_RESOURCES CONFIGURE_DEPENDS resources/toolbar/*.hvif resources/toolbar/*.png)
    set_property(TARGET Summit APPEND PROPERTY LINK_DEPENDS ${SUMMIT_TOOLBAR_RESOURCES})
    file(GLOB SUMMIT_READER_RESOURCES CONFIGURE_DEPENDS resources/reader/*.js)
    set_property(TARGET Summit APPEND PROPERTY LINK_DEPENDS ${SUMMIT_READER_RESOURCES})
    add_custom_command(TARGET Summit POST_BUILD
        COMMAND ${HAIKU_RC} -I ${CMAKE_SOURCE_DIR}/resources -o $<TARGET_FILE_DIR:Summit>/Summit.rsrc ${CMAKE_SOURCE_DIR}/resources/Summit.rdef
        COMMAND ${HAIKU_XRES} -o $<TARGET_FILE:Summit> $<TARGET_FILE_DIR:Summit>/Summit.rsrc
        COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:Summit>/resources
        COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_SOURCE_DIR}/resources/start.html $<TARGET_FILE_DIR:Summit>/resources/start.html)
endif()
